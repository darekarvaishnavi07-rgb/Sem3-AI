 - 8 Puzzle Problem using Best First Search and Python

Develop a Python program to solve the following 8-puzzle problem using Greedy Best-First Search. Use a suitable heuristic function to guide the search.

import heapq

start = (1, 2, 3,
         5, 0, 6,
         4, 7, 8)

goal = (1, 2, 3,
        4, 5, 6,
        7, 8, 0)

def h(state):
    return sum(state[i] != goal[i] and state[i] !=0
               for i in range(9))

def solve(start):
    q = [(h(start), start)]
    visited = set()

    while q:
        _, state = heapq.heappop(q)

        if state in visited:
            continue
        visited.add(state)

        print(state[:3])
        print(state[3:6])
        print(state[6:])
        print()

        if state == goal:
            print("Goal reached!")
            return

        z = state.index(0)

        for move in [-3, 3, -1, 1]:
            n = z + move

            if   0<= n < 9:
                if abs(move) == 1 and z // 3 != n // 3:
                    continue

                new = list(state)
                new[z], new[n] = new[n], new[z]
                new = tuple(new)

                if new not in visited:
                    heapq.heappush(q, (h(new), new))

solve(start)
